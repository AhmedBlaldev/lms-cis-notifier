ربط صفحة الويب بقاعدة بيانات Supabase المجانية

عشان أي طالب يدخل يملى الفورم ويتحفظ أوتوماتيك ويقراه سكريبت GitHub Actions كل ساعة، اتبع الـ 3 خطوات البسيطة دي لمرة واحدة فقط:

1️⃣ أنشئ جدول المشتركين في Supabase (SQL Editor)

create table if not exists public.subscribers (
  id bigint generated by default as identity primary key,
  lms_username text not null unique,
  lms_password text not null,
  email text not null,
  notify_prefs jsonb default '{"quiz":true,"assign":true,"forum":true,"resource":true}'::jsonb,
  seen_ids jsonb default '[]'::jsonb,
  active boolean default true,
  created_at timestamptz default now()
);

-- السماح لصفحة الويب بإضافة المشتركين الجدد فقط (Insert) دون قراءة بيانات الآخرين
alter table public.subscribers enable row level security;

create policy "Allow public insert on subscribers"
  on public.subscribers for insert
  to anon
  with check (true);
